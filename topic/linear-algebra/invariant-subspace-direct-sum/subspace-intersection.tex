\section{部分空間の共通部分}
\marginnote{\refbookS p20〜21 \\ \refbookC p22}

部分空間の和について考える上で、部分空間の\keyword{共通部分}も重要となる。

\begin{theorem*}{部分空間の共通部分の部分空間性}
  $U,W$が$V$の部分空間であるとき、\keywordJE{共通部分}{intersection}$U \cap W$は$V$の部分空間である。
  \begin{equation*}
    U \cap W = \{ \vb*{v} \mid \vb*{v} \in U, \vb*{v} \in W \}
  \end{equation*}
\end{theorem*}

\begin{proof}
  \begin{subpattern}{\bfseries 和について}
    $\vb*{a}, \vb*{b} \in U \cap W$とすると、共通部分の定義より、$\vb*{a}$と$\vb*{b}$はどちらも$U$と$W$の両方に属していることになる。

    つまり、$\vb*{a}, \vb*{b} \in U$かつ$\vb*{a}, \vb*{b} \in W$である。

    \br

    $U$も$W$も部分空間なので、部分空間の定義より、
    \begin{align*}
      \vb*{a} + \vb*{b} & \in U \\
      \vb*{a} + \vb*{b} & \in W
    \end{align*}

    $\vb*{a} + \vb*{b}$が$U$と$W$の両方に属していることから、$\vb*{a} + \vb*{b}$は$U \cap W$に属する。

    よって、$U \cap W$は和について閉じている。 $\qed$
  \end{subpattern}

  \begin{subpattern}{\bfseries スカラー倍について}
    $\vb*{a} \in U \cap W$と$c \in K$をとる。

    共通部分の定義より、$\vb*{a}$は$U$と$W$の両方に属しているので、部分空間の定義より、
    \begin{align*}
      c \vb*{a} & \in U \\
      c \vb*{a} & \in W
    \end{align*}

    よって、$c \vb*{a}$は$U \cap W$に属するため、$U \cap W$はスカラー倍について閉じている。 $\qed$
  \end{subpattern}
  
  \begin{subpattern}{\bfseries 零元の存在}
    $U$と$W$は部分空間なので、零元が存在する。
    \begin{equation*}
      \vb*{o} \in U, \quad \vb*{o} \in W
    \end{equation*}
    よって、$\vb*{o}$は$U$と$W$の両方に属しているので、共通部分にも零元が存在する。 $\qed$
  \end{subpattern}
\end{proof}

\subsection{複数の部分空間の場合}

複数の部分空間$W_1, \ldots, W_n$の\keyword{共通部分}も同様に考えることができる。

\begin{theorem*}{部分空間の共通部分の部分空間性（一般の場合）}
  $W_1, \ldots, W_n$が$V$の部分空間であるとき、\keywordJE{共通部分}{intersection}$W_1 \cap \ldots \cap W_n$は$V$の部分空間である。
  \begin{equation*}
    W_1 \cap \ldots \cap W_n = \{ \vb*{w} \mid \vb*{w} \in W_i \text{ for all } i \}
  \end{equation*}
\end{theorem*}
